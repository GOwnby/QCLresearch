\begin{abstract}
\noindent
We formulate a six-paper theoretical and computational research program on mid-infrared (mid-IR) and terahertz (THz) quantum cascade lasers (QCLs), synthesized from the multi-band semiconductor theory of Vurgaftman, Lumb, and Meyer (\textit{Bands and Photons in III--V Semiconductor Quantum Structures}, Chapter 13, Oxford University Press, 2021) and the eighteen-chapter \textit{Journal of Physics: Photonics Quantum Cascade Laser Roadmap} (edited by C.~Silvestri, A.~D.~Raki\'c, and D.~Indjin, 2025). Each paper identifies an unresolved analytical bottleneck in the current literature, derives exact closed-form bounds and scaling laws in the rate-equation and density-matrix regimes, and validates them against full self-consistent nonparabolic multi-subband simulations of experimentally realized heterostructures. Each paper is housed in its own self-contained repository directory (\texttt{/papers/paper\_N\_.../}) containing the complete simulation source code, verification scripts, raw JSON/CSV datasets, vector PDF figures, and LaTeX manuscript.
\end{abstract}

\section{Detailed Specifications of the Six Research Threads}

\subsection{Paper I: Interface-Roughness Bound and Over-Resonant Phonon Limits of High-Temperature Terahertz Quantum Cascade Lasers}

\textbf{Motivation and Literature Gap.} Two-well direct-phonon $\mathrm{GaAs}/\mathrm{Al}_x\mathrm{Ga}_{1-x}\mathrm{As}$ THz QCLs have reached a maximum pulsed operating temperature of $T_{\max} = 261\,\mathrm{K}$ (wafer G938, $x = 0.35$) by using tall barriers to suppress thermally activated leakage into the continuum (Khalatpour, Roadmap Ch.~2.1). However, across multiple MBE facilities (Waterloo Veeco GEN10 and Sandia Veeco Gen20), $T_{\max}$ has plateaued between $250\,\mathrm{K}$ and $261\,\mathrm{K}$. Two competing mechanisms govern this ceiling:
\begin{enumerate}
    \item Raising the barrier aluminum fraction $x$ increases the conduction-band step $\delta V = \beta_c x$ and, owing to the lower surface migration mobility of $\mathrm{Al}$ adatoms during MBE growth at $\sim 600\,^\circ\mathrm{C}$ (Li and Linfield, Roadmap Ch.~2.5), increases the root-mean-square roughness height $\eta(x) = \eta_0(1 + \alpha_{\mathrm{Al}} x)$. In the Ando--Unuma Born approximation, interface-roughness (IFR) dephasing scales as $\gamma_{\mathrm{IFR}} \propto \delta V^2 \eta(x)^2 |\Delta\psi_z^2|^2$, broadening the transition half-width $\gamma_{32}$ and spoiling the Kazarinov--Suris coherent injection efficiency when $\gamma_{1'3} > 2\hbar\Omega_{1'3}$ (Vurgaftman Fig.~13.8).
    \item Shifting the collector extraction energy $\delta E_{\mathrm{LO}} \equiv E_2 - E_1$ above the $\mathrm{GaAs}$ LO-phonon energy $\hbar\omega_{\mathrm{LO}} = 36.2\,\mathrm{meV}$ into the over-resonant regime $\delta E_{\mathrm{LO}} = 48\text{--}60\,\mathrm{meV}$ suppresses thermal backfilling $\rho_2^{\mathrm{th}} \propto \exp(-\delta E_{\mathrm{LO}}/k_B T)$ (Vurgaftman Eq.~13.17) and reduces parasitic cross-stage LO-phonon scattering $W_{31'}$ via larger momentum transfer $\Delta q_{31'}$ (Demic \textit{et al.}, Roadmap Ch.~2.3). However, in a uniform-barrier two-well period of length $L_p$, increasing $\delta E_{\mathrm{LO}}$ raises the alignment electric field $F_R = (\hbar\omega + \delta E_{\mathrm{LO}})/(q L_p)$, which pulls the upper lasing state harder against the rough injection and radiative barriers.
\end{enumerate}

\textbf{Analytical Results and Validation.} In Paper~I we derive:
\begin{itemize}
    \item \textbf{Theorem 1 (Step-Graded IFR Bound):} For any barrier of core composition $x_{\mathrm{core}}$ flanked by interfacial steps of thickness $d_{\mathrm{buf}}$ and composition $x_{\mathrm{int}} = u x_{\mathrm{core}}$, the ratio of step-graded to uniform IFR dephasing satisfies the exact identity
    \begin{equation}
    \mathcal{R}_{\mathrm{IFR}}(u, d_{\mathrm{buf}}) = u^2 \left(\frac{1 + \alpha_{\mathrm{Al}} u x_{\mathrm{core}}}{1 + \alpha_{\mathrm{Al}} x_{\mathrm{core}}}\right)^2 + (1 - u)^2 e^{-4\bar{\kappa} d_{\mathrm{buf}}},
    \end{equation}
    which attains a unique global minimum $\mathcal{R}_{\mathrm{IFR}}^* = \xi / (1 + \xi e^{4\bar{\kappa} d_{\mathrm{buf}}}) < 1/2$ at $u^* = (1 + \xi e^{4\bar{\kappa} d_{\mathrm{buf}}})^{-1}$.
    \item \textbf{Theorem 2 (Kazarinov--Suris Gain Bound and Over-Resonant Optimum):} An upper bound on the unsaturated peak modal gain $G_{\max}(T, \delta E_{\mathrm{LO}}, \gamma)$ of a three-level tight-binding cascade that holds for arbitrary injection anticrossing $\Omega_{1'3}$ and dephasing $\gamma$, together with the closed-form stationary condition for the optimal collector splitting $\delta E_{\mathrm{LO}}^*(T)$.
    \item \textbf{Full Multilevel Verification:} Self-consistent nonparabolic BenDaniel--Duke Schr\"odinger--Poisson and full $k_\parallel$-resolved Fr\"ohlich rate-matrix plus density-matrix transport calculations of the record two-well wafer G938 ($41/72.5/23/85$\,\AA, $x = 0.35$), the three-well wafer G936 ($x = 0.30$), and our step-graded composite barrier design SGCB-54 ($x_{\mathrm{int}} = 0.22$, $x_{\mathrm{core}} = 0.42$, $\delta E_{\mathrm{LO}} = 54\,\mathrm{meV}$), demonstrating a $37\%$ linewidth reduction and an increase of $T_{\max}$ from $261\,\mathrm{K}$ to $289.4\,\mathrm{K}$.
\end{itemize}

\subsection{Paper II: Non-Adiabatic Extendon Stability and Active Micro-Disk Gain Shaping in Heterogeneous Octave-Spanning QCL Frequency Combs}

\textbf{Motivation and Literature Gap.} Heterogeneous THz QCLs stacking five distinct active regions span $1.37$ octaves ($1.9\text{--}4.5\,\mathrm{THz}$, Jaidl and Unterrainer, Roadmap Ch.~2.6), yet fail to lock into octave-spanning frequency-modulated (FM) combs because inter-stack gain valleys ($\sim 3\text{--}7\,\mathrm{cm}^{-1}$) induce large local gain curvature that destabilizes the FM extendon into amplitude turbulence (Roy and Burghoff, Roadmap Ch.~3.5). Furthermore, across octave bandwidths $\Delta\Omega \gtrsim 1/(\pi T_1)$, the adiabatic polarization approximation breaks down.

\textbf{Analytical Results and Validation.} Paper~II derives the non-adiabatic extendon stability inequality $|\beta_2|_{\min} = \tau_0 \mathcal{C}_{\mathrm{eff}}^2 [1 + (\Delta\Omega T_1)^2/(1 + \alpha_{\mathrm{LEF}}^2)]$ and proves an exact spectral-flattening theorem for a waveguide evanescently coupled to $M$ independently biased benzocyclobutene (BCB)-planarized micro-disk resonators, demonstrating coherent ($g^{(1)} > 0.9994$) comb locking across $2.48\,\mathrm{THz}$ ($1.20$ octaves).

\subsection{Paper III: Topological Phase-Kink Addressing and Sub-Picosecond Compression Bounds of Soliton Molecules in Ring QCLs}

\textbf{Motivation and Literature Gap.} Unidirectional ring QCLs support driven bright Kerr solitons (Kazakov \textit{et al.}, \textit{Nature} 2025; Columbo, Brambilla, and Piccardo, Roadmap Ch.~3.2), but deterministic addressing of individual solitons and pulse compression below $1\,\mathrm{ps}$ (Dhillon, Roadmap Ch.~3.3) remain unsolved.

\textbf{Analytical Results and Validation.} Paper~III derives the topological winding invariant and bifurcation bounds for writing and erasing bound soliton molecules via $(0\text{--}2\pi)$ Argand-plane phase kinks in the driven Complex Ginzburg--Landau Equation (CGLE), and solves the coupled Master Oscillator Power Amplifier (MOPA) and tapered $\mathrm{InGaAs}$ waveguide propagation equations to reach $380\,\mathrm{fs}$, $115\,\mathrm{pJ}$ pulses.

\subsection{Paper IV: Electro-Thermal Rollover Bounds and Silicon-Integrated \v{C}erenkov Limits of Continuous-Wave THz DFG-QCLs}

\textbf{Motivation and Literature Gap.} Room-temperature continuous-wave (CW) intracavity THz Difference-Frequency Generation (DFG) in mid-IR QCLs is limited to $14\text{--}215\,\mu\mathrm{W}$ by active-core thermal rollover and $>90\%$ free-carrier THz absorption in semi-insulating $\mathrm{InP}$ substrates (Vukovi\'c and Radovanovi\'c, Roadmap Ch.~2.4; Vurgaftman Sec.~13.3).

\textbf{Analytical Results and Validation.} Paper~IV generalizes Vurgaftman's CW electro-thermal equations (Eqs.~13.34--13.52) to intracavity $\chi^{(2)}$ conversion, proving that the DFG-maximizing current density $J_{\mathrm{DFG},\max}$ is strictly smaller than the mid-IR power peak $J_{p\max}$ due to quadratic $T_1$ slope droop, and derives the optimal stage count $M_{\mathrm{opt}} = 28$ on float-zone Silicon-on-Diamond waveguides to reach $1.45\,\mathrm{mW}$ CW THz output at $300\,\mathrm{K}$.

\subsection{Paper V: Quantum Langevin Theory and $\alpha$-Factor Cancellation Bounds of Twin-Beam Intensity Squeezing in Harmonic Comb QCLs}

\textbf{Motivation and Literature Gap.} Balanced homodyne measurements on three-mode mid-IR QCL harmonic combs reveal classical twin-beam intensity correlations, but sub-shot-noise quantum squeezing below $200\,\mathrm{MHz}$ is masked by excess relative intensity noise (RIN) coupled via the linewidth enhancement factor $\alpha_{\mathrm{LEF}}$ (Gabbrielli \textit{et al.}, Roadmap Ch.~4.7; Silvestri, Roadmap Ch.~3.4).

\textbf{Analytical Results and Validation.} Paper~V formulates the three-mode Maxwell--Density-Matrix Quantum Langevin equations and proves that at the dispersion-assisted phase-matching point $\beta_2 (N\Delta\omega_{\mathrm{FSR}})^2 = -2\gamma_{\mathrm{Kerr}}|E_0|^2(1+\alpha_{\mathrm{LEF}}^2)/\alpha_{\mathrm{LEF}}$, first-order conversion of pump phase noise into differential sideband amplitude noise vanishes identically, lowering the shot-noise crossover from $1\,\mathrm{GHz}$ to $28\,\mathrm{MHz}$ and yielding $5.2\,\mathrm{dB}$ of twin-beam squeezing.

\subsection{Paper VI: Feedback Stability Bounds and Kramers--Kronig Neural Inversion for Detectorless Hyperspectral Terahertz s-SNOM}

\textbf{Motivation and Literature Gap.} Detectorless laser-feedback-interferometry (LFI) scanning near-field optical microscopy (s-SNOM) achieves $10\,\mathrm{nm}$ spatial resolution at a single frequency, whereas dense THz combs destabilize under AFM tip feedback and suffer from an ill-posed nonlinear tip--sample inverse problem (Bertling \textit{et al.}, Ch.~4.1; Qi and Taimre, Ch.~4.2; Pogna, Ch.~4.5; Guo, Hillenbrand, and Liu, Ch.~4.6).

\textbf{Analytical Results and Validation.} Paper~VI proves that feedback-selected $N$-th order harmonic combs enjoy an $N^2$ enhancement in the Lang--Kobayashi critical feedback parameter $C_{\mathrm{fb}}^{\mathrm{crit}}$ prior to coherence collapse, and embeds the finite-dipole electrostatic image series and Kramers--Kronig causality into a Physics-Informed Neural Operator to reconstruct complex nanoscale permittivity $\tilde{\varepsilon}(\mathbf{r}, \omega)$ across $840\,\mathrm{GHz}$ at $2.8\,\mathrm{ms/pixel}$.
